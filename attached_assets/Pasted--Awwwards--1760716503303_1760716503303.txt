5) Кнопки и карточки — применить токены

Файл: apps/web/src/components/Button.tsx — НОВЫЙ (ПОЛНЫЙ)

import React from 'react';

type Props = React.ButtonHTMLAttributes<HTMLButtonElement> & {
  variant?: 'gradient' | 'outline';
};
export default function Button({ variant='gradient', className='', ...rest }: Props) {
  if (variant === 'outline') {
    return <button {...rest} className={`btn-outline ${className}`} />;
  }
  return <button {...rest} className={`btn-gradient ${className}`} />;
}
