function generateReferralCode() {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  let result = 'PATH_'; // Ваш префикс
  for (let i = 0; i < 6; i++) {
    result += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return result; // Результат: "PATH_A3X9K2"
}

3️⃣ ПОЛУЧЕНИЕ РЕФЕРАЛЬНОГО КОДА ИЗ TELEGRAM
Frontend (App.tsx) - 4 метода обнаружения:

useEffect(() => {
  const handleReferral = async () => {
    let referralCode = null;
    
    // Метод 1: Telegram WebApp startParam
    if (window.Telegram?.WebApp?.startParam) {
      referralCode = window.Telegram.WebApp.startParam;
    }
    
    // Метод 2: Telegram initDataUnsafe
    else if (window.Telegram?.WebApp?.initDataUnsafe?.start_param) {
      referralCode = window.Telegram.WebApp.initDataUnsafe.start_param;
    }
    
    // Метод 3: URL Hash (#tgWebAppStartParam=CODE)
    else {
      const hash = window.location.hash.slice(1);
      const hashParams = new URLSearchParams(hash);
      referralCode = hashParams.get('tgWebAppStartParam');
    }
    
    // Метод 4: URL Search Parameters (?startapp=CODE)
    if (!referralCode) {
      const urlParams = new URLSearchParams(window.location.search);
      referralCode = urlParams.get('startapp') || 
                     urlParams.get('ref') || 
                     urlParams.get('referral');
    }
    
    if (referralCode) {
      await applyReferralCode(referralCode);
    }
  };
  
  handleReferral();
}, []);
4️⃣ BACKEND API ENDPOINTS
A) Применить реферальный код к существующему юзеру:
app.post("/api/users/apply-referral", async (req, res) => {
  const { userId, referralCode } = req.body;
  
  // 1. Найти пользователя
  const user = await storage.getUser(userId);
  if (!user) return res.status(404).json({ error: "User not found" });
  
  // 2. Проверить, что уже не реферил
  if (user.referredBy) {
    return res.status(400).json({ error: "Already referred" });
  }
  
  // 3. Найти реферера по коду
  const referrer = await storage.getUserByReferralCode(referralCode);
  if (!referrer) {
    return res.status(400).json({ error: "Invalid code" });
  }
  
  // 4. Нельзя реферить себя
  if (referrer.id === userId) {
    return res.status(400).json({ error: "Cannot refer yourself" });
  }
  
  // 5. Обновить пользователя (связать с рефералом)
  await storage.updateUser(user.id, { referredBy: referrer.id });
  
  // 6. Увеличить счётчик рефералов у реферера
  await storage.updateUserReferrals(referrer.id, referrer.referrals + 1);
  
  // 7. Создать запись о награде для реферера
  await storage.createReferralReward({
    referrerId: referrer.id,
    referredId: user.id,
    rewardAmount: 50,
    rewardType: "signup",
    claimed: false
  });
  
  res.json({ success: true, rewardAmount: 50 });
});
B) Забрать накопленные награды:
app.post("/api/users/:userId/referrals/claim", async (req, res) => {
  const { userId } = req.params;
  
  // 1. Получить все награды пользователя
  const rewards = await storage.getReferralRewards(userId);
  const unclaimed = rewards.filter(r => !r.claimed);
  
  if (unclaimed.length === 0) {
    return res.json({ claimedAmount: 0 });
  }
  
  // 2. Посчитать сумму
  const totalReward = unclaimed.reduce((sum, r) => sum + r.rewardAmount, 0);
  
  // 3. Пометить как забранные
  const rewardIds = unclaimed.map(r => r.id);
  await storage.updateReferralRewardsClaimed(rewardIds);
  
  res.json({ 
    claimedAmount: totalReward,
    rewardsCount: unclaimed.length 
  });
});
C) Получить статистику рефералов:
app.get("/api/users/:userId/referrals", async (req, res) => {
  const user = await storage.getUser(req.params.userId);
  const referredUsers = await storage.getUserReferrals(user.id);
  const rewards = await storage.getReferralRewards(user.id);
  
  res.json({
    referralCode: user.referralCode,
    totalReferrals: user.referrals,
    referredUsers: referredUsers.map(u => ({
      id: u.id,
      username: u.username,
      createdAt: u.createdAt
    })),
    rewards: rewards,
    totalEarnedFromReferrals: rewards.reduce((sum, r) => sum + r.rewardAmount, 0)
  });
});

6️⃣ FLOW ДИАГРАММА
1. Пользователь A получает код "PATH_XYZ123"
2. Пользователь A делится ссылкой: t.me/bot?startapp=PATH_XYZ123
3. Пользователь B кликает по ссылке
4. Telegram открывает Mini App с параметром startParam="PATH_XYZ123"
5. Frontend извлекает код из window.Telegram.WebApp.startParam
6. Frontend вызывает /api/users/apply-referral
7. Backend:
   - Находит реферера по коду
   - Связывает B с A (referredBy)
   - Увеличивает счётчик A (referrals++)
   - Создаёт награду для A (50 токенов)
8. Пользователь A может забрать награду через /api/users/:userId/referrals/claim
7️⃣ ВАЖНЫЕ ПРОВЕРКИ
✅ Проверка на дубликаты: Пользователь может быть реферён только один раз
✅ Проверка на self-referral: Нельзя использовать свой собственный код
✅ Уникальность кодов: Каждый код генерируется с проверкой в БД
✅ Claimed/Unclaimed: Награды можно забрать только один раз
