### 1) apps/api/src/auth/tgLoginVerify.ts (НОВЫЙ, ПОЛНЫЙ)
```ts
import crypto from "crypto";
import type { Request, Response } from "express";
import { prisma } from "../prisma";
import { signAccessToken, signRefreshToken } from "../auth/jwt"; // уже существуют
import { z } from "zod";

const env = {
  BOT_TOKEN: process.env.TELEGRAM_BOT_TOKEN || "",
  ALLOWED_SKEW: Number(process.env.LOGIN_ALLOWED_SKEW_SECONDS || "86400"),
};

const Input = z.object({
  id: z.number(),
  first_name: z.string().optional(),
  last_name: z.string().optional(),
  username: z.string().optional(),
  photo_url: z.string().optional(),
  auth_date: z.string(), // unix ts в секундах
  hash: z.string(),
});

function checkTelegramLogin(data: Record<string, string>) {
  // По спецификации: secret_key = SHA256(bot_token)
  const secretKey = crypto.createHash("sha256").update(env.BOT_TOKEN).digest();
  // Собрать data_check_string: отсортированные ключи, кроме hash
  const fields = Object.keys(data)
    .filter((k) => k !== "hash")
    .sort()
    .map((k) => `${k}=${data[k]}`)
    .join("\n");
  const hmac = crypto.createHmac("sha256", secretKey).update(fields).digest("hex");
  return hmac === data.hash;
}

export async function tgLoginVerifyHandler(req: Request, res: Response) {
  const parsed = Input.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ ok: false, error: "BAD_INPUT" });
  }
  const payload = parsed.data;

  // Проверка времени
  const nowSec = Math.floor(Date.now() / 1000);
  const authSec = Number(payload.auth_date);
  if (!Number.isFinite(authSec) || Math.abs(nowSec - authSec) > env.ALLOWED_SKEW) {
    return res.status(401).json({ ok: false, error: "AUTH_EXPIRED" });
  }

  // Проверка подписи
  const valid = checkTelegramLogin(
    Object.fromEntries(Object.entries(payload).map(([k, v]) => [k, String(v)]))
  );
  if (!valid) {
    return res.status(401).json({ ok: false, error: "INVALID_HASH" });
  }

  // Создаём/обновляем пользователя по tgId
  const tgId = String(payload.id);
  const name = [payload.first_name, payload.last_name].filter(Boolean).join(" ") || payload.username || "User";
  const user = await prisma.user.upsert({
    where: { tgId },
    update: {
      username: payload.username ?? undefined,
      name,
      // не перетираем пол/дату — они задаются в регистрации Mini App
    },
    create: {
      tgId,
      username: payload.username ?? null,
      name,
      gender: "other",
      age: 18,
      birthdayDate: new Date(0), // заглушка-дату потом исправит регистрация
      timezone: "Europe/Moscow",
      energy: 10,
      energyResetAt: new Date(),
      referralCode: `R${Math.random().toString(36).slice(2,8).toUpperCase()}`,
    },
  });

  const access = signAccessToken({ uid: user.id, tgId });
  const refresh = signRefreshToken({ uid: user.id, tgId });

  return res.json({
    ok: true,
    data: {
      accessToken: access,
      refreshToken: refresh,
      user: { id: user.id, tgId: user.tgId, name: user.name, username: user.username },
    },
  });
}

Копировать код
import React, { useEffect, useRef } from "react";
import { http } from "../lib/api"; // обёртка с baseURL

declare global {
  interface Window {
    TelegramLoginWidget?: any;
  }
}

export default function Login() {
  const containerRef = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    if (!containerRef.current) return;

    // Создаём виджет через data-атрибуты — безопасно и просто
    const div = document.createElement("div");
    div.setAttribute("id", "tg-login");
    div.setAttribute("class", "flex justify-center");
    div.setAttribute("data-telegram-login", import.meta.env.VITE_TG_BOT_USERNAME || "astro_orb_bot");
    div.setAttribute("data-size", "large");
    div.setAttribute("data-radius", "12");
    div.setAttribute("data-request-access", "write");
    // Коллбек в JS — передаём данные в наш API
    (window as any).onTelegramAuth = async (user: any) => {
      try {
        const { data } = await http.post("/auth/tg-login", user);
        if (data?.ok) {
          localStorage.setItem("accessToken", data.data.accessToken);
          localStorage.setItem("refreshToken", data.data.refreshToken);
          // редирект в Mini App/личный кабинет
          window.location.href = "/"; 
        } else {
          alert("Ошибка входа: " + (data?.error || "unknown"));
        }
      } catch (e: any) {
        alert("Ошибка сети при входе");
      }
    };
    div.setAttribute("data-onauth", "onTelegramAuth(user)");

    containerRef.current.appendChild(div);

    // Если скрипт подгружен позже — он сам инициализирует виджет по data-атрибутам
  }, []);

  return (
    <div className="w-full h-screen flex items-center justify-center">
      <div ref={containerRef} />
    </div>
  );
}

README.md — ДОБАВИТЬ раздел «Telegram Login Widget (веб-логин)»
Вставить:

Зачем: веб-логин на вашем домене, удобный для лендингов, десктопа и deep-link из соцсетей.

Как настроить у @BotFather:

/setdomain → укажите точный домен (совпадает с PUBLIC_WEB_ORIGIN), обязательно HTTPS.
Пример: app.example.com.

/setmenu (опционально) → добавьте кнопку Web App, если хотите вход из меню бота.

Проверьте CSP/Headers на фронте: не блокируйте https://telegram.org/js/telegram-widget.js.

Как проверить:

открыть ${PUBLIC_WEB_ORIGIN}/login, нажать «Войти с Telegram», в консоли сервера видеть POST /auth/tg-login → user+JWT.

Убедиться, что после логина локально сохранены токены и происходит редирект.
