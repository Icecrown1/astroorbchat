from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from datetime import datetime
from .important_dates_api import calculate_natal_chart, calculate_positions
from .db import db, solar_returns
from .energy import spend_sphere
import swisseph as swe

router = APIRouter(prefix="/solar", tags=["solar"])

class SolarRequest(BaseModel):
    target_year: int
    method: str = "exact_return"  # "exact_return" | "same_date"
    lat: float
    lon: float
    tz_offset_minutes: int
    place_label: str

@router.post("/compute")
def compute_solar(req: SolarRequest, user=Depends(auth_user)):
    natal = db.get_natal_chart(user.id)
    if not natal or not natal["time_iso"] or not natal["lat"]:
        raise HTTPException(status_code=409, detail="NATAL_INCOMPLETE")

    existing = db.query(solar_returns).where(
        (solar_returns.user_id == user.id) &
        (solar_returns.target_year == req.target_year) &
        (solar_returns.lat == req.lat) &
        (solar_returns.lon == req.lon) &
        (solar_returns.method == req.method)
    ).first()

    if existing:
        return {"ok": True, "data": existing.data}

    # 1. Получаем натальную долготу Солнца
    natal_sun = natal["planets"]["Sun"]["longitude"]

    # 2. Метод exact_return — поиск момента возвращения Солнца
    if req.method == "exact_return":
        jd_start = swe.julday(req.target_year, natal["month"], natal["day"], natal["hour"])
        jd = jd_start
        for _ in range(30):  # до 30 итераций
            lon_sun, _ = swe.calc_ut(jd, swe.SUN)
            diff = (lon_sun - natal_sun + 360) % 360
            if diff < 0.001:
                break
            jd -= diff / 0.9856  # грубая коррекция
        utc_dt = swe.revjul(jd, gregflag=1)
        exact_moment = datetime(*utc_dt)
    else:
        # 3. Метод same_date — просто тот же день/время в новом году
        y, m, d = req.target_year, natal["month"], natal["day"]
        h, mi, s = map(int, natal["time_iso"].split(":"))
        exact_moment = datetime(y, m, d, h, mi, s)

    # 4. Строим солярную карту
    solar_chart = calculate_natal_chart(
        date_iso=exact_moment.date().isoformat(),
        time_iso=exact_moment.time().isoformat(),
        lat=req.lat,
        lon=req.lon,
        tz_offset_minutes=req.tz_offset_minutes
    )

    db.insert(solar_returns).values(
        user_id=user.id,
        target_year=req.target_year,
        method=req.method,
        latitude=req.lat,
        longitude=req.lon,
        tz_offset_minutes=req.tz_offset_minutes,
        place_label=req.place_label,
        exact_moment_utc=exact_moment,
        data=solar_chart
    ).execute()

    # Списываем 1 сферу
    spend_sphere(user.id, 1, "solar_return")

    return {"ok": True, "data": solar_chart}
