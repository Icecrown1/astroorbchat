  | "Jupiter" | "Saturn" | "Uranus" | "Neptune" | "Pluto";

export type PlanetIconProps = {
  name: PlanetName;
  size?: number;            // px, по умолчанию 32
  variant?: "gold" | "bronze" | "ink"; // цветовая тема
  title?: string;           // для a11y
  className?: string;
  animated?: boolean;       // включает лёгкую анимацию
};

const PALETTE = {
  gold:   { fill: "#cfa24b", stroke: "#8d6a25" },
  bronze: { fill: "#b87436", stroke: "#7a4d22" },
  ink:    { fill: "#444",    stroke: "#222"    },
};

function useColors(variant: PlanetIconProps["variant"]) {
  return PALETTE[variant ?? "gold"];
}

export default function PlanetIcon({
  name,
  size = 32,
  variant = "gold",
  title,
  className,
  animated = true,
}: PlanetIconProps) {
  const { fill, stroke } = useColors(variant);
  const common = {
    width: size,
    height: size,
    viewBox: "0 0 64 64",
    role: "img",
    "aria-label": title ?? name,
  } as const;

  const animClass = animated ? "po-anim" : "";

  switch (name) {
    case "Sun":
      return (
        <svg {...common} className={`${className ?? ""} ${animClass} po-sun`}>
          <title>{title ?? "Sun"}</title>
          <defs>
            <radialGradient id="gSun" cx="50%" cy="50%" r="50%">
              <stop offset="0%" stopColor="#ffe08a" />
              <stop offset="60%" stopColor={fill} />
              <stop offset="100%" stopColor={stroke} />
            </radialGradient>
          </defs>
          <circle cx="32" cy="32" r="16" className="po-sun-core" fill="url(#gSun)" stroke={stroke} strokeWidth="1.5"/>
          {Array.from({length:12}).map((_,i)=> {
            const a = (i * Math.PI*2)/12;
            const x1 = 32 + Math.cos(a)*22, y1 = 32 + Math.sin(a)*22;
            const x2 = 32 + Math.cos(a)*30, y2 = 32 + Math.sin(a)*30;
            return <line key={i} x1={x1} y1={y1} x2={x2} y2={y2} className="po-sun-ray" stroke={fill} strokeWidth="2" strokeLinecap="round"/>;
          })}
        </svg>

    case "Saturn":

/* Чёткий рендер svg */
svg { shape-rendering: geometricPrecision; }

/* Анимации. Почти незаметные, но оживляют UI */
@keyframes poPulse {
  0% { transform: scale(1); opacity: 1; }
  50% { transform: scale(1.04); opacity: .95; }
  100% { transform: scale(1); opacity: 1; }
}

@keyframes poSunRay {
  0% { opacity: .85; }
  50% { opacity: .45; }
  100% { opacity: .85; }
}

@keyframes poMoonShimmer {
  0% { filter: brightness(100%); }
  50% { filter: brightness(112%); }
  100% { filter: brightness(100%); }
}

@keyframes poSaturnRings {
  0%   { transform: rotate(0deg); }
  100% { transform: rotate(6deg); } /* очень медленное покачивание */
}

/* Глобальный переключатель анимаций */
.po-anim .po-pulse {
  transform-origin: center;
  animation: poPulse 3.6s ease-in-out infinite;
}
.po-anim.po-sun .po-sun-ray {
  animation: poSunRay 2.8s ease-in-out infinite;
}
.po-anim.po-moon .po-moon-crescent {
  transform-origin: 32px 32px;
  animation: poMoonShimmer 4.2s ease-in-out infinite;
}
.po-anim.po-saturn .po-saturn-rings {
  transform-origin: 32px 34px;
  animation: poSaturnRings 14s linear infinite alternate;
}

/* Уважение к пользователям, отключившим анимации */
@media (prefers-reduced-motion: reduce) {
  .po-anim .po-pulse,
  .po-anim.po-sun .po-sun-ray,
  .po-anim.po-moon .po-moon-crescent,
  .po-anim.po-saturn .po-saturn-rings {
    animation: none !important;
  }
}

/* Фокус и доступность */
button:focus-visible { outline: 2px solid rgba(207,162,75,.6); outline-offset: 2px; }
